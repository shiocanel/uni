CREATE DATABASE BankManagement;
GO

USE BankManagement;
GO

CREATE TABLE Branches(
    BranchID INT PRIMARY KEY IDENTITY(1, 1),
    BranchName NVARCHAR(100) NOT NULL,
    BranchCode VARCHAR(10) UNIQUE NOT NULL,
    BranchAddress NVARCHAR(255) NOT NULL,
    City NVARCHAR(100) NOT NULL,
    PhoneNumber VARCHAR(20)
);

CREATE TABLE AccountTypes(
    AccountTypeID INT PRIMARY KEY IDENTITY(1, 1),
    TypeName VARCHAR(50) UNIQUE NOT NULL,
    InterestRate DECIMAL(5, 4) DEFAULT 0.00 CHECK (InterestRate >= 0),
    MonthlyFee DECIMAL(10, 2) DEFAULT 0.00
);

CREATE TABLE Customers(
    CustomerID INT PRIMARY KEY IDENTITY(1, 1),
    FirstName NVARCHAR(100) NOT NULL,
    LastName NVARCHAR(100) NOT NULL,
    PersonalNumericCode VARCHAR(20) UNIQUE NOT NULL,
    DateOfBirth DATE NOT NULL CHECK (DateOfBirth <= GETDATE()),
    Email VARCHAR(100) UNIQUE,
    PhoneNumber VARCHAR(20) NOT NULL,
    CustomerAddress NVARCHAR(255)
);

CREATE TABLE Employees(
    EmployeeID INT IDENTITY(1,1) PRIMARY KEY,
    BranchID INT NOT NULL,
    FirstName NVARCHAR(100) NOT NULL,
    LastName NVARCHAR(100) NOT NULL,
    JobTitle VARCHAR(50) NOT NULL,
    HireDate DATE DEFAULT GETDATE(),
    Salary DECIMAL(10,2) NOT NULL CHECK (Salary > 0),
    CONSTRAINT FK_Employees_Branches FOREIGN KEY (BranchID) REFERENCES Branches(BranchID)
);

CREATE TABLE Accounts (
    AccountID INT IDENTITY(1,1) PRIMARY KEY,
    BranchID INT NOT NULL,
    AccountTypeID INT NOT NULL,
    AccountNumber VARCHAR(30) UNIQUE NOT NULL,
    Balance DECIMAL(18,2) DEFAULT 0.00,
    Currency VARCHAR(3) DEFAULT 'USD',
    AccountStatus VARCHAR(20) DEFAULT 'Active' CHECK (AccountStatus IN ('Active', 'Suspended', 'Closed')),
    OpenedDate DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_Accounts_Branches FOREIGN KEY (BranchID) REFERENCES Branches(BranchID),
    CONSTRAINT FK_Accounts_AccountTypes FOREIGN KEY (AccountTypeID) REFERENCES AccountTypes(AccountTypeID)
);

CREATE TABLE CustomerAccounts(
    CustomerID INT NOT NULL,
    AccountID INT NOT NULL,
    AccountRole VARCHAR(30) DEFAULT 'Primary' CHECK (AccountRole IN ('Primary', 'Secondary', 'Guarantor')),
    LinkDate DATETIME DEFAULT GETDATE(),
    PRIMARY KEY (CustomerID, AccountID),
    CONSTRAINT FK_CustomerAccounts_Customers FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID),
    CONSTRAINT FK_CustomerAccounts_Accounts FOREIGN KEY (AccountID) REFERENCES Accounts(AccountID)
);

CREATE TABLE Transactions (
    TransactionID INT IDENTITY(1,1) PRIMARY KEY,
    AccountID INT NOT NULL,
    TransactionType VARCHAR(20) NOT NULL CHECK (TransactionType IN ('Deposit', 'Withdrawal', 'Transfer', 'Fee')),
    Amount DECIMAL(18,2) NOT NULL CHECK (Amount > 0),
    TransactionDate DATETIME DEFAULT GETDATE(),
    Description NVARCHAR(255),
    CONSTRAINT FK_Transactions_Accounts FOREIGN KEY (AccountID) REFERENCES Accounts(AccountID)
);

CREATE TABLE Cards (
    CardID INT IDENTITY(1,1) PRIMARY KEY,
    AccountID INT NOT NULL,
    CardNumber VARCHAR(16) UNIQUE NOT NULL,
    CardType VARCHAR(20) NOT NULL CHECK (CardType IN ('Debit', 'Credit')),
    ExpiryDate DATE NOT NULL,
    CVV VARCHAR(4) NOT NULL,
    IsActive BIT DEFAULT 1,
    CONSTRAINT FK_Cards_Accounts FOREIGN KEY (AccountID) REFERENCES Accounts(AccountID)
);

CREATE TABLE Loans (
    LoanID INT IDENTITY(1,1) PRIMARY KEY,
    CustomerID INT NOT NULL,
    BranchID INT NOT NULL,
    LoanAmount DECIMAL(18,2) NOT NULL CHECK (LoanAmount > 0),
    InterestRate DECIMAL(5,4) NOT NULL,
    TermInMonths INT NOT NULL CHECK (TermInMonths > 0),
    StartDate DATE DEFAULT GETDATE(),
    LoanStatus VARCHAR(20) DEFAULT 'Pending' CHECK (LoanStatus IN ('Pending', 'Approved', 'PaidOff', 'Defaulted')),
    CONSTRAINT FK_Loans_Customers FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID),
    CONSTRAINT FK_Loans_Branches FOREIGN KEY (BranchID) REFERENCES Branches(BranchID)
);

CREATE TABLE LoanPayments (
    PaymentID INT IDENTITY(1,1) PRIMARY KEY,
    LoanID INT NOT NULL,
    PaymentAmount DECIMAL(18,2) NOT NULL CHECK (PaymentAmount > 0),
    PaymentDate DATETIME DEFAULT GETDATE(),
    PrincipalAmount DECIMAL(18,2) NOT NULL,
    InterestAmount DECIMAL(18,2) NOT NULL,
    CONSTRAINT FK_LoanPayments_Loans FOREIGN KEY (LoanID) REFERENCES Loans(LoanID)
);USE BankManagement;

SELECT c.FirstName, c.LastName, c.Email, a.AccountNumber, a.Balance
FROM Customers c
JOIN CustomerAccounts ca ON c.CustomerID = ca.CustomerID
JOIN Accounts a ON ca.AccountID = a.AccountID
WHERE c.CustomerAddress LIKE '%Bucuresti%';


--tranzactii > 500 m - n
SELECT c.FirstName, c.LastName, a.AccountNumber, t.TransactionType, t.Amount, t.TransactionDate
FROM Transactions t
JOIN Accounts a ON t.AccountID = a.AccountID
JOIN CustomerAccounts ca ON a.AccountID = ca.AccountID
JOIN Customers c ON ca.CustomerID = c.CustomerID
WHERE t.Amount > 500.00;


--carduri care expira dupa 2026
SELECT cr.CardNumber, cr.CardType, cr.ExpiryDate, a.AccountNumber, b.BranchName
FROM Cards cr
JOIN Accounts a ON cr.AccountID = a.AccountID
JOIN Branches b ON a.BranchID = b.BranchID
WHERE cr.IsActive = 1 AND cr.ExpiryDate > '2026-12-31';


--conturi active cu balance > 1000 
SELECT DISTINCT at.TypeName, at.InterestRate
FROM AccountTypes at
JOIN Accounts a ON at.AccountTypeID = a.AccountTypeID
WHERE a.AccountStatus = 'Active' AND a.Balance > 1000.00;


--employees din Cluj-Napoca cu salary > 3000 
SELECT e.FirstName, e.LastName, e.JobTitle, e.Salary, b.BranchName, b.City
FROM Employees e
JOIN Branches b ON e.BranchID = b.BranchID
WHERE b.City = 'Cluj-Napoca' AND e.Salary > 3000.00;


--total balance grupate by branch si account type
SELECT b.BranchName, at.TypeName, SUM(a.Balance) AS TotalBalance, COUNT(a.AccountID) AS TotalAccountsCount
FROM Accounts a
JOIN Branches b ON a.BranchID = b.BranchID
JOIN AccountTypes at ON a.AccountTypeID = at.AccountTypeID
GROUP BY b.BranchName, at.TypeName;


--branches cu suma loan > 50000
SELECT b.BranchName, COUNT(l.LoanID) AS NumberOfLoans, SUM(l.LoanAmount) AS TotalLoanPortfolio
FROM Loans l
JOIN Branches b ON l.BranchID = b.BranchID
JOIN Customers c ON l.CustomerID = c.CustomerID
GROUP BY b.BranchName
HAVING SUM(l.LoanAmount) > 50000.00;


--cccount types si branches unde media soldului > 5000
SELECT at.TypeName, b.BranchName, AVG(a.Balance) AS AverageBalance
FROM Accounts a
JOIN AccountTypes at ON a.AccountTypeID = at.AccountTypeID
JOIN Branches b ON a.BranchID = b.BranchID
GROUP BY at.TypeName, b.BranchName
HAVING AVG(a.Balance) > 5000.00;

-- tabelul de versionare
IF OBJECT_ID('DatabaseVersion', 'U') IS NOT NULL
    DROP TABLE DatabaseVersion;
GO

CREATE TABLE DatabaseVersion (
    VersionID INT PRIMARY KEY
);

INSERT INTO DatabaseVersion (VersionID) VALUES 
(0);
GO

-- modificarea tipului unei coloane
GO
CREATE PROCEDURE sp_UpgradeToV1 AS
BEGIN
    ALTER TABLE Branches ALTER COLUMN PhoneNumber VARCHAR(30);
    PRINT 'Upgraded to V1: Modified Branches.PhoneNumber to VARCHAR(30).';
END;
GO

CREATE PROCEDURE sp_DowngradeToV0 AS
BEGIN
    ALTER TABLE Branches ALTER COLUMN PhoneNumber VARCHAR(20);
    PRINT 'Downgraded to V0: Reverted Branches.PhoneNumber to VARCHAR(20).';
END;
GO


-- adauga constrangere
CREATE PROCEDURE sp_UpgradeToV2 AS
BEGIN
    ALTER TABLE Customers ADD CONSTRAINT DF_Customers_CustomerAddress DEFAULT 'Not Provided' FOR CustomerAddress;
    PRINT 'Upgraded to V2: Added DEFAULT constraint on Customers.CustomerAddress.';
END;
GO

CREATE PROCEDURE sp_DowngradeToV1 AS
BEGIN
    ALTER TABLE Customers DROP CONSTRAINT DF_Customers_CustomerAddress;
    PRINT 'Downgraded to V1: Dropped DEFAULT constraint on Customers.CustomerAddress.';
END;
GO


-- crearea stergere auditlogs
CREATE PROCEDURE sp_UpgradeToV3 AS
BEGIN
    CREATE TABLE AuditLogs (
        LogID INT IDENTITY(1,1) PRIMARY KEY,
        ActionDescription NVARCHAR(255) NOT NULL,
        LogDate DATETIME DEFAULT GETDATE()
    );
    PRINT 'Upgraded to V3: Created table AuditLogs.';
END;
GO

CREATE PROCEDURE sp_DowngradeToV2 AS
BEGIN
    DROP TABLE AuditLogs;
    PRINT 'Downgraded to V2: Dropped table AuditLogs.';
END;
GO


-- adauga camp nou
CREATE PROCEDURE sp_UpgradeToV4 AS
BEGIN
    ALTER TABLE Branches ADD BranchRating DECIMAL(3,2) NULL;
    PRINT 'Upgraded to V4: Added column BranchRating to Branches.';
END;
GO

CREATE PROCEDURE sp_DowngradeToV3 AS
BEGIN
    ALTER TABLE Branches DROP COLUMN BranchRating;
    PRINT 'Downgraded to V3: Dropped column BranchRating from Branches.';
END;
GO


--constrangere pe o cheie straine
CREATE PROCEDURE sp_UpgradeToV5 AS
BEGIN
    ALTER TABLE Cards DROP CONSTRAINT FK_Cards_Accounts;
    PRINT 'Upgraded to V5: Dropped foreign key constraint FK_Cards_Accounts.';
END;
GO

CREATE PROCEDURE sp_DowngradeToV4 AS
BEGIN
    ALTER TABLE Cards ADD CONSTRAINT FK_Cards_Accounts FOREIGN KEY (AccountID) REFERENCES Accounts(AccountID);
    PRINT 'Downgraded to V4: Re-created foreign key constraint FK_Cards_Accounts.';
END;
GO


-- procedura de tranzatie intre versiuni
GO
CREATE PROCEDURE sp_ChangeDatabaseVersion
    @TargetVersion INT
AS
BEGIN
    DECLARE @CurrentVersion INT;
    

    SELECT @CurrentVersion = VersionID 
    FROM DatabaseVersion;

    IF @TargetVersion < 0 OR @TargetVersion > 5
    BEGIN
        RAISERROR('Invalid target version!', 16, 1);
        RETURN;
    END

    WHILE @CurrentVersion < @TargetVersion
    BEGIN
        IF @CurrentVersion = 0 
        BEGIN 
            EXEC sp_UpgradeToV1; 
            UPDATE DatabaseVersion SET VersionID = 1; 
        END
        ELSE IF @CurrentVersion = 1 
        BEGIN 
            EXEC sp_UpgradeToV2; 
            UPDATE DatabaseVersion SET VersionID = 2; 
        END
        ELSE IF @CurrentVersion = 2 
        BEGIN 
            EXEC sp_UpgradeToV3; 
            UPDATE DatabaseVersion SET VersionID = 3; 
        END
        ELSE IF @CurrentVersion = 3 
        BEGIN 
            EXEC sp_UpgradeToV4; 
            UPDATE DatabaseVersion SET VersionID = 4; 
        END
        ELSE IF @CurrentVersion = 4 
        BEGIN 
            EXEC sp_UpgradeToV5; 
            UPDATE DatabaseVersion SET VersionID = 5; 
        END
        
        SELECT @CurrentVersion = VersionID 
        FROM DatabaseVersion;
    END

    WHILE @CurrentVersion > @TargetVersion
    BEGIN
        IF @CurrentVersion = 5 
        BEGIN 
            EXEC sp_DowngradeToV4; 
            UPDATE DatabaseVersion SET VersionID = 4; 
        END
        ELSE IF @CurrentVersion = 4 
        BEGIN 
            EXEC sp_DowngradeToV3; 
            UPDATE DatabaseVersion SET VersionID = 3; 
        END
        ELSE IF @CurrentVersion = 3 
        BEGIN 
            EXEC sp_DowngradeToV2; 
            UPDATE DatabaseVersion SET VersionID = 2; 
        END
        ELSE IF @CurrentVersion = 2 
        BEGIN 
            EXEC sp_DowngradeToV1; 
            UPDATE DatabaseVersion SET VersionID = 1; 
        END
        ELSE IF @CurrentVersion = 1 
        BEGIN 
            EXEC sp_DowngradeToV0; 
            UPDATE DatabaseVersion SET VersionID = 0; 
        END
        
        SELECT @CurrentVersion = VersionID FROM DatabaseVersion;
    END

    PRINT 'Database to version: ' + CAST(@TargetVersion AS VARCHAR(10));
END;
GO
